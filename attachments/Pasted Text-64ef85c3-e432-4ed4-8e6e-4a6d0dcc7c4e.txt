==================================================
1) Miles-from-shop delivery zone
==================================================

Add to ShopSettingsPublic in src/lib/shop-types.ts:

  deliveryZoneMode: "paint" | "radius";
  deliveryRadiusMiles: number;

Defaults: mode "paint" (keep current shops working), radius 5.

DB (same alter-if-not-exists pattern already used in shop-server.ts):

  alter table shop_settings add column if not exists delivery_zone_mode text not null default 'paint';
  alter table shop_settings add column if not exists delivery_radius_miles numeric not null default 5;

Wire them through publicSettings() and saveShopSettings() (deliveryZoneMode / deliveryRadiusMiles).
Clamp radius to 0.5–30 miles.

In src/lib/geo.ts add:

  export const SHOP_LAT = MAP_CENTER[0];
  export const SHOP_LNG = MAP_CENTER[1];

  export function milesBetween(lat1, lng1, lat2, lng2) {
    // haversine, Earth radius 3958.8 miles
  }

  export function inRadius(lat, lng, miles) {
    return milesBetween(SHOP_LAT, SHOP_LNG, lat, lng) <= miles;
  }

  export function isAddressDeliverable(opts: {
    mode: "paint" | "radius";
    radiusMiles: number;
    cells: string[];
    lat: number;
    lng: number;
    query?: string;
    label?: string;
    city?: string;
    zip?: string;
  }) {
    if (opts.mode === "radius") {
      return Number.isFinite(opts.lat) && Number.isFinite(opts.lng)
        && inRadius(opts.lat, opts.lng, opts.radiusMiles);
    }
    if (isNorthfieldDelivery(opts)) return false;
    return opts.cells.length > 0 && cellSetHas(opts.cells, opts.lat, opts.lng);
  }

Use isAddressDeliverable in:

Radius mode ignores the Northfield hard-block. Distance is the only rule.

When mode is paint, hasZones stays "cells.length > 0".
